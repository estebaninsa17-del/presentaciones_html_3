4. Funciones utilitarias 
-- Hash SHA-256 para contraseñas
CREATE OR REPLACE FUNCTION seguridad.fn_hashpassword(p_password TEXT)
RETURNS TEXT LANGUAGE sql AS $$
    SELECT LOWER(ENCODE(DIGEST(p_password, 'sha256'), 'hex'));
$$;

-- Total certificados por voluntario
CREATE OR REPLACE FUNCTION voluntariado.fn_totalcertificados(p_idvoluntario INT)
RETURNS INT LANGUAGE plpgsql AS $$
BEGIN
    RETURN (SELECT COUNT(*) FROM voluntariado.certificados
            WHERE voluntarios_idvoluntarios = p_idvoluntario);
END;
$$;

-- Total donado por usuario
CREATE OR REPLACE FUNCTION donaciones.fn_totaldonado(p_idusuario INT)
RETURNS INT LANGUAGE plpgsql AS $$
BEGIN
    RETURN COALESCE((SELECT SUM(monto) FROM donaciones.donaciones
                    WHERE usuarios_idusuarios = p_idusuario), 0);
END;
$$;

-- Promedio global donaciones confirmadas
CREATE OR REPLACE FUNCTION donaciones.fn_promediodonaciones()
RETURNS NUMERIC(12,2) LANGUAGE plpgsql AS $$
BEGIN
    RETURN (SELECT AVG(monto) FROM donaciones.donaciones
            WHERE estadopago = 'Confirmada');
END;
$$;

-- Usuarios activos totales
CREATE OR REPLACE FUNCTION usuarios.fn_usuariosactivos()
RETURNS INT LANGUAGE plpgsql AS $$
BEGIN
    RETURN (SELECT COUNT(*) FROM usuarios.usuarios
            WHERE estadodecuenta = 'Activo');
END;
$$;

-- Cupos disponibles reales de una ubicación
CREATE OR REPLACE FUNCTION ubicaciones.fn_cuposdisponibles(p_idubicacion INT)
RETURNS INT LANGUAGE plpgsql AS $$
DECLARE
    v_cupos     INT;
    v_inscritos INT;
BEGIN
    SELECT CAST(cuposdisponibles AS INT) INTO v_cupos
    FROM ubicaciones.ubicaciones WHERE idubicaciones = p_idubicacion;

    SELECT COUNT(*) INTO v_inscritos
    FROM voluntariado.actividades a
    JOIN voluntariado.postulaciones p ON a.idactividades = p.actividades_idactividades
    WHERE a.ubicaciones_idubicaciones = p_idubicacion
      AND p.estadopostulacion = 'Aprobada';

    RETURN GREATEST(v_cupos - v_inscritos, 0);
END;
$$;

5. Trigger encriptación de contraseñas 
CREATE OR REPLACE FUNCTION usuarios.fn_encriptar_contrasena()
RETURNS TRIGGER LANGUAGE plpgsql AS $$
BEGIN
    IF NEW.contrasena IS NOT NULL
       AND NOT (LENGTH(NEW.contrasena) = 64 AND NEW.contrasena ~ '^[a-f0-9]+$') THEN
        NEW.contrasena := seguridad.fn_hashpassword(NEW.contrasena);
    END IF;
    RETURN NEW;
END;
$$;

CREATE TRIGGER trg_encriptar_contrasena
    BEFORE INSERT OR UPDATE OF contrasena
    ON usuarios.usuarios
    FOR EACH ROW EXECUTE FUNCTION usuarios.fn_encriptar_contrasena();

6. Triggers de auditoría automática 
CREATE OR REPLACE FUNCTION seguridad.fn_registrar_auditoria()
RETURNS TRIGGER LANGUAGE plpgsql AS $$
DECLARE v_desc TEXT;
BEGIN
    v_desc := CASE TG_OP
        WHEN 'INSERT' THEN 'Registro insertado en ' || TG_TABLE_SCHEMA || '.' || TG_TABLE_NAME
        WHEN 'UPDATE' THEN 'Registro actualizado en ' || TG_TABLE_SCHEMA || '.' || TG_TABLE_NAME
        WHEN 'DELETE' THEN 'Registro eliminado en ' || TG_TABLE_SCHEMA || '.' || TG_TABLE_NAME
    END;
    INSERT INTO seguridad.auditoria (idusuario, accion, esquema, tabla, fecha, descripcion)
    VALUES (NULL, TG_OP, TG_TABLE_SCHEMA, TG_TABLE_NAME, NOW(), v_desc);
    RETURN NULL;
END;
$$;

7. Trigger: generar detalle al confirmar donación 
CREATE OR REPLACE FUNCTION donaciones.fn_generar_detalle_donacion()
RETURNS TRIGGER LANGUAGE plpgsql AS $$
BEGIN
    IF NEW.estadopago = 'Confirmada' AND
       (OLD IS NULL OR OLD.estadopago <> 'Confirmada') THEN
        INSERT INTO donaciones.detalle_donacion (fechaderegistro, totaldonado, donaciones_iddonaciones)
        VALUES (CURRENT_DATE, NEW.monto, NEW.iddonaciones);
    END IF;
    RETURN NEW;
END;
$$;

CREATE TRIGGER trg_detalle_donacion
    AFTER INSERT OR UPDATE OF estadopago ON donaciones.donaciones
    FOR EACH ROW EXECUTE FUNCTION donaciones.fn_generar_detalle_donacion();

8. Trigger: generar comprobante al confirmar 
CREATE OR REPLACE FUNCTION donaciones.fn_generar_comprobante()
RETURNS TRIGGER LANGUAGE plpgsql AS $$
DECLARE v_codigo INT;
BEGIN
    IF NEW.estadopago = 'Confirmada' AND
       (OLD IS NULL OR OLD.estadopago <> 'Confirmada') THEN

        v_codigo := (EXTRACT(EPOCH FROM NOW()) * 100)::BIGINT % 2147483647;

        WHILE EXISTS (
            SELECT 1 FROM donaciones.comprobantes WHERE codigocomprobante = v_codigo
        ) LOOP
            v_codigo := v_codigo + 1;
        END LOOP;

        INSERT INTO donaciones.comprobantes (codigocomprobante, fechaexpedicion, donaciones_iddonaciones)
        VALUES (v_codigo, CURRENT_DATE, NEW.iddonaciones);
    END IF;
    RETURN NEW;
END;
$$;

CREATE TRIGGER trg_comprobante_donacion
    AFTER INSERT OR UPDATE OF estadopago ON donaciones.donaciones
    FOR EACH ROW EXECUTE FUNCTION donaciones.fn_generar_comprobante();

9. Vistas 
-- Voluntarios activos
CREATE OR REPLACE VIEW voluntariado.vw_voluntariosactivos AS
SELECT v.idvoluntarios, u.nombrecompleto, u.correo, u.estadodecuenta, u.fecharegisto
FROM voluntariado.voluntarios v
JOIN usuarios.usuarios u ON v.usuarios_idusuarios = u.idusuarios
WHERE u.estadodecuenta = 'Activo';

-- Actividades con su ubicación
CREATE OR REPLACE VIEW voluntariado.vw_actividadesconubicacion AS
SELECT a.idactividades, a.nombreactividad, a.descripcion, a.fechainicio, a.fechafin,
       ub.nombreubicacion, ub.direccionubicacion, ub.cuposdisponibles
FROM voluntariado.actividades a
JOIN ubicaciones.ubicaciones ub ON a.ubicaciones_idubicaciones = ub.idubicaciones;

-- Postulaciones detalladas
CREATE OR REPLACE VIEW voluntariado.vw_postulaciones AS
SELECT p.idpostulaciones, u.nombrecompleto AS voluntario, a.nombreactividad,
       p.fechapostulacion, p.estadopostulacion, p.comentario, p.diasespera
FROM voluntariado.postulaciones p
JOIN voluntariado.voluntarios v ON p.voluntarios_idvoluntarios = v.idvoluntarios
JOIN usuarios.usuarios u        ON v.usuarios_idusuarios = u.idusuarios
JOIN voluntariado.actividades a ON p.actividades_idactividades = a.idactividades;

10. Stored Procedures 
-- Registrar nueva donación
CREATE OR REPLACE PROCEDURE donaciones.sp_registrar_donacion(
    p_idusuario  INT,
    p_iddonante  INT,
    p_monto      INT,
    p_metodo     VARCHAR,
    p_anonima    VARCHAR DEFAULT 'No'
)
LANGUAGE plpgsql AS $$
BEGIN
    INSERT INTO donaciones.donaciones
        (fechadonacion, monto, estadopago, metodopago, anonima, usuarios_idusuarios, donante_idinvitados)
    VALUES
        (CURRENT_DATE, p_monto, 'Pendiente', p_metodo, p_anonima, p_idusuario, p_iddonante);
END;
$$;

-- Aprobar postulación
CREATE OR REPLACE PROCEDURE voluntariado.sp_aprobar_postulacion(p_idpostulacion INT)
LANGUAGE plpgsql AS $$
BEGIN
    UPDATE voluntariado.postulaciones
    SET estadopostulacion = 'Aprobada'
    WHERE idpostulaciones = p_idpostulacion
      AND estadopostulacion = 'Pendiente';

    IF NOT FOUND THEN
        RAISE EXCEPTION 'Postulación % no encontrada o ya procesada', p_idpostulacion;
    END IF;
END;
$$;
