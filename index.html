2. Tablas orden según dependencias FK 
-- ─── SEGURIDAD (sin dependencias externas — va primero) ───

CREATE TABLE seguridad.auditoria (
    idauditoria  SERIAL       PRIMARY KEY,
    idusuario    INT          NULL,
    accion       VARCHAR(50)  NOT NULL,
    esquema      VARCHAR(50)  NOT NULL,
    tabla        VARCHAR(100) NOT NULL,
    fecha        TIMESTAMP    NOT NULL DEFAULT NOW(),
    descripcion  TEXT         NULL,

    CONSTRAINT ck_audit_accion CHECK (accion IN ('INSERT','UPDATE','DELETE','LOGIN','LOGOUT','EXPORT'))
);
-- ─── 🟣 MÓDULO UBICACIONES ───

CREATE TABLE ubicaciones.ubicaciones (
    idubicaciones        SERIAL      PRIMARY KEY,
    nombreubicacion      VARCHAR(45) NOT NULL,
    direccionubicacion   VARCHAR(45) NOT NULL,
    cuposdisponibles     VARCHAR(45) NOT NULL,

    CONSTRAINT ck_ubicacion_nombre    CHECK (LENGTH(TRIM(nombreubicacion)) >= 2),
    CONSTRAINT ck_ubicacion_direccion CHECK (LENGTH(TRIM(direccionubicacion)) >= 3)
);
-- ─── 🟡 MÓDULO USUARIOS ───

CREATE TABLE usuarios.usuarios (
    idusuarios       SERIAL       PRIMARY KEY,
    nombrecompleto   VARCHAR(45)  NOT NULL,
    numerodocumento  BIGINT       NOT NULL UNIQUE,
    tipodocumento    VARCHAR(45)  NOT NULL,
    correo           VARCHAR(45)  NOT NULL UNIQUE,
    contrasena       VARCHAR(255) NOT NULL,
    telefono         BIGINT       NULL,
    direccion        VARCHAR(45)  NULL,
    ciudad           VARCHAR(45)  NULL,
    estadodecuenta   VARCHAR(45)  NOT NULL DEFAULT 'Activo',
    fecharegisto     DATE         NOT NULL DEFAULT CURRENT_DATE,

    CONSTRAINT ck_usuario_tipodocumento CHECK (tipodocumento IN ('CC','TI','CE','PA')),
    CONSTRAINT ck_usuario_contrasena    CHECK (LENGTH(contrasena) >= 8),
    CONSTRAINT ck_usuario_correo        CHECK (correo LIKE '%_@__%.__%'),
    CONSTRAINT ck_usuario_estadocuenta  CHECK (estadodecuenta IN ('Activo','Inactivo','Suspendido')),
    CONSTRAINT ck_usuario_fecharegisto  CHECK (fecharegisto >= '2000-01-01' AND fecharegisto <= CURRENT_DATE)
);

CREATE TABLE usuarios.roles (
    idroles    SERIAL      PRIMARY KEY,
    nombrerol  VARCHAR(45) NOT NULL UNIQUE,
    CONSTRAINT ck_rol_nombre CHECK (LENGTH(TRIM(nombrerol)) >= 2)
);

CREATE TABLE usuarios.roles_has_usuarios (
    roles_idroles       INT NOT NULL,
    usuarios_idusuarios INT NOT NULL,
    PRIMARY KEY (roles_idroles, usuarios_idusuarios),
    CONSTRAINT fk_rhu_rol     FOREIGN KEY (roles_idroles)       REFERENCES usuarios.roles    (idroles),
    CONSTRAINT fk_rhu_usuario FOREIGN KEY (usuarios_idusuarios) REFERENCES usuarios.usuarios (idusuarios)
);

CREATE TABLE usuarios.notificaciones (
    idnotificaciones    SERIAL      PRIMARY KEY,
    mensaje             VARCHAR(45) NOT NULL,
    fechaenvio          DATE        NOT NULL DEFAULT CURRENT_DATE,
    usuarios_idusuarios INT         NOT NULL,
    CONSTRAINT fk_notif_usuario FOREIGN KEY (usuarios_idusuarios) REFERENCES usuarios.usuarios (idusuarios),
    CONSTRAINT ck_notif_fecha   CHECK (fechaenvio >= '2000-01-01' AND fechaenvio <= CURRENT_DATE)
);
-- ─── 🔴 MÓDULO COORDINACIÓN ───

CREATE TABLE coordinacion.coordinadores (
    idcoordinadores     SERIAL      PRIMARY KEY,
    arearesponsable     VARCHAR(45) NOT NULL,
    usuarios_idusuarios INT         NOT NULL UNIQUE,
    CONSTRAINT fk_coord_usuario FOREIGN KEY (usuarios_idusuarios) REFERENCES usuarios.usuarios (idusuarios),
    CONSTRAINT ck_coord_area    CHECK (LENGTH(TRIM(arearesponsable)) >= 3)
);
-- ─── 🟢 MÓDULO DONACIONES ───

CREATE TABLE donaciones.donante (
    idinvitados         SERIAL      PRIMARY KEY,
    propositodonacion   VARCHAR(45) NOT NULL,
    usuarios_idusuarios INT         NOT NULL,
    CONSTRAINT fk_donante_usuario FOREIGN KEY (usuarios_idusuarios) REFERENCES usuarios.usuarios (idusuarios),
    CONSTRAINT ck_donante_prop    CHECK (LENGTH(TRIM(propositodonacion)) >= 3)
);

CREATE TABLE donaciones.donaciones (
    iddonaciones        SERIAL      PRIMARY KEY,
    fechadonacion       DATE        NOT NULL DEFAULT CURRENT_DATE,
    monto               INT         NOT NULL,
    estadopago          VARCHAR(45) NOT NULL DEFAULT 'Pendiente',
    metodopago          VARCHAR(45) NOT NULL,
    anonima             VARCHAR(45) NOT NULL DEFAULT 'No',
    usuarios_idusuarios INT         NOT NULL,
    donante_idinvitados INT         NOT NULL,

    CONSTRAINT fk_don_usuario FOREIGN KEY (usuarios_idusuarios) REFERENCES usuarios.usuarios  (idusuarios),
    CONSTRAINT fk_don_donante FOREIGN KEY (donante_idinvitados) REFERENCES donaciones.donante (idinvitados),
    CONSTRAINT ck_don_monto   CHECK (monto > 0),
    CONSTRAINT ck_don_estado  CHECK (estadopago IN ('Pendiente','Confirmada','Rechazada','Anulada')),
    CONSTRAINT ck_don_anonima CHECK (anonima IN ('Si','No')),
    CONSTRAINT ck_don_metodo  CHECK (metodopago IN ('Efectivo','Transferencia','Tarjeta Débito','Tarjeta Crédito','PSE','Nequi','Daviplata')),
    CONSTRAINT ck_don_fecha   CHECK (fechadonacion >= '2000-01-01' AND fechadonacion <= CURRENT_DATE)
);

CREATE TABLE donaciones.detalle_donacion (
    idhistorialdonaciones   SERIAL PRIMARY KEY,
    fechaderegistro         DATE   NOT NULL DEFAULT CURRENT_DATE,
    totaldonado             INT    NOT NULL,
    donaciones_iddonaciones INT    NOT NULL,
    CONSTRAINT fk_det_donacion FOREIGN KEY (donaciones_iddonaciones) REFERENCES donaciones.donaciones (iddonaciones),
    CONSTRAINT ck_det_total    CHECK (totaldonado >= 0),
    CONSTRAINT ck_det_fecha    CHECK (fechaderegistro >= '2000-01-01' AND fechaderegistro <= CURRENT_DATE)
);

CREATE TABLE donaciones.comprobantes (
    idcomprobantes          SERIAL PRIMARY KEY,
    codigocomprobante       INT    NOT NULL UNIQUE,
    fechaexpedicion         DATE   NOT NULL DEFAULT CURRENT_DATE,
    donaciones_iddonaciones INT    NOT NULL,
    CONSTRAINT fk_comp_donacion FOREIGN KEY (donaciones_iddonaciones) REFERENCES donaciones.donaciones (iddonaciones),
    CONSTRAINT ck_comp_fecha    CHECK (fechaexpedicion >= '2000-01-01' AND fechaexpedicion <= CURRENT_DATE)
);

CREATE TABLE donaciones.metodo_pago (
    idmetodo_pago SERIAL PRIMARY KEY
);

CREATE TABLE donaciones.donaciones_has_metodo_pago (
    donaciones_iddonaciones INT NOT NULL,
    metodo_pago_idmetodo    INT NOT NULL,
    PRIMARY KEY (donaciones_iddonaciones, metodo_pago_idmetodo),
    CONSTRAINT fk_dhmp_donacion FOREIGN KEY (donaciones_iddonaciones) REFERENCES donaciones.donaciones (iddonaciones),
    CONSTRAINT fk_dhmp_metodo   FOREIGN KEY (metodo_pago_idmetodo)   REFERENCES donaciones.metodo_pago (idmetodo_pago)
);
-- ─── 🔵 MÓDULO VOLUNTARIADO ───

CREATE TABLE voluntariado.voluntarios (
    idvoluntarios           SERIAL PRIMARY KEY,
    usuarios_idusuarios     INT    NOT NULL UNIQUE,
    usuarios_idusuarios_ref INT    NOT NULL,
    CONSTRAINT fk_vol_usuario FOREIGN KEY (usuarios_idusuarios) REFERENCES usuarios.usuarios (idusuarios)
);

CREATE TABLE voluntariado.actividades (
    idactividades             SERIAL      PRIMARY KEY,
    nombreactividad           VARCHAR(45) NOT NULL,
    descripcion               VARCHAR(45) NULL,
    fechainicio               DATE        NOT NULL,
    fechafin                  DATE        NULL,
    ubicaciones_idubicaciones INT         NOT NULL,
    CONSTRAINT fk_act_ubicacion   FOREIGN KEY (ubicaciones_idubicaciones) REFERENCES ubicaciones.ubicaciones (idubicaciones),
    CONSTRAINT ck_act_nombre      CHECK (LENGTH(TRIM(nombreactividad)) >= 3),
    CONSTRAINT ck_act_fechainicio CHECK (fechainicio >= '2000-01-01'),
    CONSTRAINT ck_act_fechafin    CHECK (fechafin IS NULL OR fechafin >= fechainicio)
);

CREATE TABLE voluntariado.postulaciones (
    idpostulaciones                 SERIAL      PRIMARY KEY,
    fechapostulacion                DATE        NOT NULL DEFAULT CURRENT_DATE,
    estadopostulacion               VARCHAR(45) NOT NULL DEFAULT 'Pendiente',
    comentario                      VARCHAR(45) NULL,
    diasespera                      INT         NULL,
    voluntarios_idvoluntarios       INT         NOT NULL,
    voluntarios_usuarios_idusuarios INT         NOT NULL,
    usuarios_idusuarios             INT         NOT NULL,
    actividades_idactividades       INT         NOT NULL,
    CONSTRAINT fk_post_voluntario FOREIGN KEY (voluntarios_idvoluntarios) REFERENCES voluntariado.voluntarios (idvoluntarios),
    CONSTRAINT fk_post_usuario    FOREIGN KEY (usuarios_idusuarios)       REFERENCES usuarios.usuarios       (idusuarios),
    CONSTRAINT fk_post_actividad  FOREIGN KEY (actividades_idactividades)  REFERENCES voluntariado.actividades  (idactividades),
    CONSTRAINT ck_post_estado     CHECK (estadopostulacion IN ('Pendiente','Aprobada','Rechazada')),
    CONSTRAINT ck_post_diasespera CHECK (diasespera >= 0)
);

CREATE TABLE voluntariado.certificados (
    idcertificados                  SERIAL      PRIMARY KEY,
    nombrevoluntario                VARCHAR(45) NOT NULL,
    actividadasociada               VARCHAR(45) NOT NULL,
    voluntarios_idvoluntarios       INT         NOT NULL,
    voluntarios_usuarios_idusuarios INT         NOT NULL,
    usuarios_idusuarios             INT         NOT NULL,
    actividades_idactividades       INT         NOT NULL,
    CONSTRAINT ck_cert_nombre CHECK (LENGTH(TRIM(nombrevoluntario)) >= 2)
);

3. Índices 
-- ubicaciones
CREATE INDEX ix_ubic_nombre      ON ubicaciones.ubicaciones (nombreubicacion);

-- usuarios
CREATE INDEX ix_usu_correo       ON usuarios.usuarios (correo);
CREATE INDEX ix_usu_documento    ON usuarios.usuarios (numerodocumento);
CREATE INDEX ix_usu_estado       ON usuarios.usuarios (estadodecuenta) INCLUDE (idusuarios, nombrecompleto);
CREATE INDEX ix_usu_fecharegisto ON usuarios.usuarios (fecharegisto DESC);
CREATE INDEX ix_notif_usuario    ON usuarios.notificaciones (usuarios_idusuarios) INCLUDE (fechaenvio);

-- voluntariado
CREATE INDEX ix_act_nombre       ON voluntariado.actividades (nombreactividad);
CREATE INDEX ix_act_fechainicio  ON voluntariado.actividades (fechainicio DESC) INCLUDE (nombreactividad);
CREATE INDEX ix_post_estado      ON voluntariado.postulaciones (estadopostulacion) WHERE estadopostulacion = 'Pendiente';

-- donaciones
CREATE INDEX ix_don_fecha        ON donaciones.donaciones (fechadonacion DESC) INCLUDE (monto, estadopago);
CREATE INDEX ix_don_estado       ON donaciones.donaciones (estadopago) INCLUDE (iddonaciones, monto);
CREATE INDEX ix_don_monto        ON donaciones.donaciones (monto DESC);

-- seguridad
CREATE INDEX ix_aud_fecha        ON seguridad.auditoria (fecha DESC);
CREATE INDEX ix_aud_tabla        ON seguridad.auditoria (tabla) INCLUDE (accion, fecha);
CREATE INDEX ix_aud_usuario      ON seguridad.auditoria (idusuario) INCLUDE (fecha, accion);
